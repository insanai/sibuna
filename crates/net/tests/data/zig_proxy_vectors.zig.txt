//! Records the bytes Zig's streamProxy puts on the wire for a set of scenarios.
//! Each scenario: a raw client request, a canned origin response; we record what the origin
//! received, what the client received, the result, the origin status and the captured head.
const std = @import("std");
const Io = std.Io;
const net = @import("net");
const io = std.testing.io;

const Mode = enum { buffered, partial, chunked_stream, chunked_complete };

const Scenario = struct {
    name: []const u8,
    client: []const u8,
    origin: []const u8,
    mode: Mode = .buffered,
    partial: usize = 0,
    scheme: []const u8 = "http",
    keep_alive: bool = true,
    to_eof: bool = false,
};

const OriginTask = struct {
    listener: *Io.net.Server,
    response: []const u8,
    to_eof: bool,
    received: [200 * 1024]u8 = undefined,
    len: usize = 0,

    fn save(task: *OriginTask, bytes: []const u8) void {
        @memcpy(task.received[0..bytes.len], bytes);
        task.len = bytes.len;
    }
};

fn contentLength(head: []const u8) ?usize {
    const at = std.mem.indexOf(u8, head, "\r\nContent-Length: ") orelse return null;
    const digits = std.mem.sliceTo(head[at + 18 ..], '\r');
    return std.fmt.parseInt(usize, digits, 10) catch null;
}

fn originMain(task: *OriginTask) void {
    const stream = task.listener.accept(io) catch return;
    defer stream.close(io);
    var buf: [200 * 1024]u8 = undefined;
    var reader = stream.reader(io, &buf);
    var wbuf: [4096]u8 = undefined;
    var writer = stream.writer(io, &wbuf);
    const r = &reader.interface;
    const head_end = while (true) {
        if (std.mem.indexOf(u8, r.buffered(), "\r\n\r\n")) |i| break i + 4;
        r.fillMore() catch {
            task.save(r.buffered());
            return;
        };
    };
    if (task.to_eof) {
        writer.interface.writeAll(task.response) catch {};
        writer.interface.flush() catch {};
        while (true) r.fillMore() catch break;
        task.save(r.buffered());
        return;
    }
    const head = r.buffered()[0..head_end];
    if (contentLength(head)) |n| {
        while (r.buffered().len < head_end + n) r.fillMore() catch break;
    } else if (std.mem.indexOf(u8, head, "Transfer-Encoding: chunked") != null) {
        while (!std.mem.endsWith(u8, r.buffered(), "\n0\r\n\r\n")) r.fillMore() catch break;
    }
    task.save(r.buffered());
    writer.interface.writeAll(task.response) catch {};
    writer.interface.flush() catch {};
}

const Capture = struct {
    bytes: [32 * 1024]u8 = undefined,
    len: usize = 0,
    seen: bool = false,

    fn call(context: *anyopaque, raw: []const u8) void {
        const self: *Capture = @ptrCast(@alignCast(context));
        @memcpy(self.bytes[0..raw.len], raw);
        self.len = raw.len;
        self.seen = true;
    }
};

var pool: net.proxy.Pool = .{};

fn hexLine(tag: []const u8, bytes: []const u8) void {
    std.debug.print("{s} ", .{tag});
    for (bytes) |b| std.debug.print("{x:0>2}", .{b});
    std.debug.print("\n", .{});
}

fn run(s: Scenario) !void {
    const addr = try Io.net.IpAddress.parse("127.0.0.1", 0);
    var origin_listener = try addr.listen(io, .{ .reuse_address = true });
    defer origin_listener.deinit(io);
    const task = try std.heap.page_allocator.create(OriginTask);
    defer std.heap.page_allocator.destroy(task);
    task.* = .{ .listener = &origin_listener, .response = s.origin, .to_eof = s.to_eof };
    const thread = try std.Thread.spawn(.{}, originMain, .{task});

    var front = try addr.listen(io, .{ .reuse_address = true });
    defer front.deinit(io);
    const client_end = try front.socket.address.connect(io, .{ .mode = .stream });
    defer client_end.close(io);
    const server_end = try front.accept(io);
    defer server_end.close(io);
    {
        var cbuf: [4096]u8 = undefined;
        var cw = client_end.writer(io, &cbuf);
        try cw.interface.writeAll(s.client);
        try cw.interface.flush();
        try client_end.shutdown(io, .send);
    }
    const conn_buf = try std.heap.page_allocator.alloc(u8, 200 * 1024);
    defer std.heap.page_allocator.free(conn_buf);
    var reader = server_end.reader(io, conn_buf);
    var wbuf: [16 * 1024]u8 = undefined;
    var writer = server_end.writer(io, &wbuf);
    const r = &reader.interface;
    while (true) r.fillMore() catch break;
    const head_len = std.mem.indexOf(u8, r.buffered(), "\r\n\r\n").? + 4;
    var req = try net.parseRequest(r.buffered()[0..head_len]);
    var decoder: net.chunked.Decoder = .{};
    var upload: net.proxy.Upload = undefined;
    switch (s.mode) {
        .buffered => {
            const n = req.contentLength() orelse 0;
            req.body = r.buffered()[head_len..][0..n];
            r.toss(head_len + n);
            upload = .declared(&req);
        },
        .partial => {
            req.body = r.buffered()[head_len..][0..s.partial];
            r.toss(head_len + s.partial);
            upload = .declared(&req);
        },
        .chunked_stream => {
            req.body = "";
            r.toss(head_len);
            upload = .{ .chunked = &decoder };
        },
        .chunked_complete => {
            const raw = r.buffered()[head_len..];
            const step = try decoder.decode(raw);
            if (!decoder.done()) return error.Incomplete;
            req.body = raw[0..step.output];
            r.toss(head_len + step.consumed);
            upload = .{ .length = step.output };
        },
    }
    var status: u16 = 0;
    var capture: Capture = .{};
    const audit = net.ProxyAudit{
        .client_ip = "203.0.113.9",
        .status = "PASS",
        .rule = "default/allow",
        .scheme = s.scheme,
        .response_status = &status,
        .response_head = .{ .context = &capture, .call = Capture.call },
    };
    const result = net.proxy.streamProxy(&pool, .{
        .io = io,
        .client = .{
            .stream = server_end,
            .reader = r,
            .writer = &writer.interface,
            .relay = .{ .idle_timeout_seconds = 5 },
        },
        .upstream_host = "127.0.0.1",
        .upstream_port = origin_listener.socket.address.ip4.port,
        .request = &req,
        .upload = upload,
        .audit = audit,
        .keep_alive = s.keep_alive,
    });
    pool.drain(io);
    server_end.shutdown(io, .both) catch {};
    // Unblocks an origin that was never contacted.
    if (origin_listener.socket.address.connect(io, .{ .mode = .stream })) |dummy| {
        dummy.close(io);
    } else |_| {}
    thread.join();

    const out = try std.heap.page_allocator.alloc(u8, 200 * 1024);
    defer std.heap.page_allocator.free(out);
    var len: usize = 0;
    var rbuf: [4096]u8 = undefined;
    var cr = client_end.reader(io, &rbuf);
    while (true) {
        const n = cr.interface.readSliceShort(out[len..]) catch break;
        if (n == 0) break;
        len += n;
    }

    std.debug.print("scenario {s}\n", .{s.name});
    hexLine("client", s.client);
    hexLine("origin", s.origin);
    std.debug.print("mode {s} {d}\n", .{ @tagName(s.mode), s.partial });
    std.debug.print("scheme {s}\n", .{s.scheme});
    std.debug.print("keep_alive {d}\n", .{@intFromBool(s.keep_alive)});
    std.debug.print("to_eof {d}\n", .{@intFromBool(s.to_eof)});
    if (result) |keep| {
        std.debug.print("result ok {d}\n", .{@intFromBool(keep)});
    } else |err| {
        std.debug.print("result err {s}\n", .{@errorName(err)});
    }
    std.debug.print("status {d}\n", .{status});
    if (capture.seen) hexLine("head", capture.bytes[0..capture.len]) else std.debug.print("head -\n", .{});
    hexLine("origin_received", task.received[0..task.len]);
    hexLine("client_received", out[0..len]);
    std.debug.print("end\n", .{});
}

const ws_head = "GET /ws HTTP/1.1\r\nHost: h\r\nConnection: keep-alive, Upgrade\r\n" ++
    "Upgrade: websocket\r\nSec-WebSocket-Key: dGhlIHNhbXBsZSBub25jZQ==\r\n" ++
    "Sec-WebSocket-Version: 13\r\nSec-WebSocket-Extensions: x\r\n";
const ws_accept = "Sec-WebSocket-Accept: s3pPLMBiTxaQ9kYGzzhZRbK+xOo=\r\n";
const chunked_upload = "POST /up HTTP/1.1\r\nHost: h\r\nTransfer-Encoding: chunked\r\n\r\n" ++
    "5;ext=1\r\nhello\r\n6\r\n world\r\n0\r\nX-T: 1\r\n\r\n";
const ok2 = "HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nok";

fn bigBody() []const u8 {
    const S = struct {
        var bytes: [300]u8 = undefined;
    };
    for (&S.bytes, 0..) |*b, i| b.* = "abcdefghijklmnopqrstuvwxyz0123456789"[i % 36];
    return &S.bytes;
}

fn bigHead() []const u8 {
    const S = struct {
        var bytes: [17 * 1024 + 64]u8 = undefined;
    };
    const prefix = "HTTP/1.1 200 OK\r\nX-Big: ";
    @memcpy(S.bytes[0..prefix.len], prefix);
    @memset(S.bytes[prefix.len .. S.bytes.len - 4], 'a');
    @memcpy(S.bytes[S.bytes.len - 4 ..], "\r\n\r\n");
    return &S.bytes;
}
